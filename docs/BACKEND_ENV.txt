RK FASHIONS — BACKEND ENVIRONMENT VARIABLES
===========================================

File: server/.env   (template: server/.env.example)
On Render: Service → Environment. Keep these values secret — never commit them.

# ─── Required ──────────────────────────────────────────────────────────
NODE_ENV=production
# Port the API listens on. Render sets this automatically — leave it out there.
PORT=5050

# MongoDB connection string (MongoDB Atlas → Connect → Drivers).
MONGODB_URI=mongodb+srv://<user>:<password>@<cluster>.mongodb.net/rk-fashions?retryWrites=true&w=majority

# Public address of the website (no trailing slash). Used for CORS, email links,
# sitemap.xml. Several addresses can be listed, separated by commas; the first
# one is used in emails.
CLIENT_URL=https://rkfashions.in

# Long random strings. Generate each with:  openssl rand -hex 32
JWT_SECRET=<random 64-character string>
ADMIN_TOKEN=<different random 64-character string>

# ─── Sign-in cookies ───────────────────────────────────────────────────
# lax  = backend serves the website (recommended, default)
# none = frontend on a different domain (e.g. Vercel + Render)
COOKIE_SAMESITE=lax

# ─── Email notifications (optional, recommended) ───────────────────────
# Order, appointment and enquiry confirmations, quotations, shipping emails,
# new-order alerts and password reset. Any SMTP provider works.
# Gmail: SMTP_HOST=smtp.gmail.com, SMTP_PORT=465, SMTP_PASS = a Google "App password".
STUDIO_EMAIL=studio@rkfashions.in
SMTP_HOST=smtp.gmail.com
SMTP_PORT=465
SMTP_USER=studio@rkfashions.in
SMTP_PASS=<app password>
MAIL_FROM="RK FASHIONS <studio@rkfashions.in>"

# ─── Image storage (optional locally, REQUIRED in production) ──────────
# Cloudinary dashboard → API Keys. Without these, uploads are saved to the
# server disk, which Render wipes on every deploy.
CLOUDINARY_CLOUD_NAME=<cloud name>
CLOUDINARY_API_KEY=<api key>
CLOUDINARY_API_SECRET=<api secret>

# ─── Payments ──────────────────────────────────────────────────────────
# Razorpay Dashboard → Account & Settings → API Keys.
# Use rzp_test_… keys while testing, rzp_live_… after account activation.
RAZORPAY_KEY_ID=rzp_test_<key id>
RAZORPAY_KEY_SECRET=<key secret>
# Razorpay Dashboard → Webhooks → https://<your-site>/api/payments/razorpay/webhook
# events: payment.captured, order.paid — paste the same secret here.
RAZORPAY_WEBHOOK_SECRET=<webhook secret>

# Cash on delivery: true to offer it, false to hide it.
COD_ENABLED=false

# Shipping charge in rupees (0 = complimentary shipping).
SHIPPING_FEE=0

# ─── Only for one-off scripts / tests (not needed on the server) ───────
# ADMIN_PASSWORD=<used by `npm run create-admin` instead of prompting>
# TEST_MONGODB_URI=mongodb://127.0.0.1:27017/rk-fashions-test
